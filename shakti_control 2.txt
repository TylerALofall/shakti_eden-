 ['◈','Reflection','sHist'],['♫','Mureka','sMur'],['⬗','Files','sFiles']];

const Q13=['Title the last 10 cycles','Summary','Did I finish — if no, the reason','Was it successful',
 'If I was to redo it, what would I do different','Tools used','What tools would have helped',
 'What could Tyler have done different to help','What files did I save, and their paths',
 'Memories recalled that helped, by the epoch of the first memory in the batch','Notes',

let S={hbOn:0,hbMin:5,hbCount:0,web:0,term:0,shot:0,goal:'',msgs:[],menu:[],notes:[],
 goals:[],files:[],blocks:[],mureka:[],termLog:[],webLog:[],page:0,loc:'',locR:''};
let timer=null,nextAt=0,rec=null,micOn=0,fr=0,ls=0;

async function load(){try{const r=await window.storage.get('shakti:v2');
  if(r&&r.value)S=Object.assign(S,JSON.parse(r.value));}catch(e){}}
async function save(){try{await window.storage.set('shakti:v2',JSON.stringify(S));}catch(e){}}
function addr(sec,ord){const e=Math.floor(Date.now()/1000);
  if(e!==ls){ls=e;fr=0;}else fr=(fr+1)%10000;
  return'['+e+':'+String(fr).padStart(4,'0')+']-['+sec+']-['+String(ord||1).padStart(2,'0')+']';}
function esc(s){return String(s).replace(/[<>&]/g,c=>({'<':'&lt;','>':'&gt;','&':'&amp;'}[c]));}

RAIL.forEach(([g,n,id])=>{const b=document.createElement('button');
  b.className='rn';b.dataset.id=id;
  b.innerHTML='<span class="aur">'+g+'</span><span class="sym">'+g+'</span>'+
              '<span class="lbl">'+n.toUpperCase()+'</span>';
  b.onclick=()=>open(id);$('#rail').appendChild(b);});
function fire(id){const b=$('.rn[data-id="'+id+'"]');if(!b)return;
  b.classList.add('fire');setTimeout(()=>b.classList.remove('fire'),620);}

function paint(m){const d=document.createElement('div');d.className='msg '+m.c;
  d.innerHTML=esc(m.t)+'<span class="a">'+m.a+'</span>';
  stream.appendChild(d);stream.scrollTop=stream.scrollHeight;}
function render(){stream.innerHTML='';S.msgs.forEach(paint);}

const ta=$('#in');
ta.oninput=()=>{ta.style.height='auto';ta.style.height=Math.min(ta.scrollHeight,104)+'px';
  $('#send').disabled=!ta.value.trim();};
ta.onkeydown=e=>{if(e.key==='Enter'&&!e.shiftKey){e.preventDefault();send();}};
$('#send').onclick=send;
function send(){const t=ta.value.trim();if(!t)return;
  ta.value='';ta.style.height='auto';$('#send').disabled=1;
  const m={c:'me',t:t,a:addr('Gh',1)};S.msgs.push(m);paint(m);save();}

function setLoc(t,r){S.loc=t;S.locR=r||'';
  $('#locT').textContent=t||'idle';$('#locR').textContent=r||'—';save();}

$('#heart').onclick=()=>open('sHb');
function hbSync(){$('#hbMin').value=S.hbMin;$('#hbCount').textContent=S.hbCount;
  $('#beats').textContent=S.hbCount+' BEATS';$('#hbGo').textContent=S.hbOn?'Stop':'Start';
  $('#hbNote').textContent=S.hbMin===0?'Zero yields. No beat is scheduled.'
    :'Due every '+S.hbMin+' min. One bounded action per beat.';}
$('#hbMin').oninput=e=>{S.hbMin=+e.target.value;hbSync();save();if(S.hbOn)arm();};
$('#hbGo').onclick=()=>{S.hbOn=!S.hbOn;S.hbOn?arm():disarm();hbSync();save();};
function arm(){clearInterval(timer);frame.classList.add('live');
  if(S.hbMin===0){nextAt=0;frame.style.setProperty('--bpm','5s');
    $('#cd').className='off';$('#cd').textContent='yield';setLoc('yielding','Gb');return;}
  frame.style.setProperty('--bpm',Math.max(1.6,Math.min(6,S.hbMin/3))+'s');
  nextAt=Date.now()+S.hbMin*60000;$('#cd').className='';
  timer=setInterval(beat,S.hbMin*60000);setLoc('waiting','Gb');}
function disarm(){clearInterval(timer);timer=null;nextAt=0;
  frame.classList.remove('live');$('#cd').className='off';$('#cd').textContent='—';setLoc('','');}
function beat(){S.hbCount++;nextAt=Date.now()+S.hbMin*60000;
  $('#hbCount').textContent=S.hbCount;$('#beats').textContent=S.hbCount+' BEATS';
  fire('sHb');setLoc('beat '+S.hbCount,'Gb');save();}
setInterval(()=>{$('#addr').textContent=addr('Ga',1);
  if(!nextAt||!S.hbOn)return;
  const d=Math.max(0,nextAt-Date.now()),m=Math.floor(d/60000),s=Math.floor(d%60000/1000);
  $('#cd').textContent=String(m).padStart(2,'0')+':'+String(s).padStart(2,'0');},1000);

function open(id){$('#veil').classList.add('open');$('#'+id).classList.add('open');
  const f={sHb:hbSync,sMenu:miRender,sNote:ntRender,sHist:pgRender,sFiles:fRender,
    sMur:muRender,sTerm:tmRender,sWeb:wbRender,sTime:tiRender,sOut:outRender,
    sGoal:()=>{$('#goalTxt').value=S.goal;goalSrc();goalList();}}[id];
  if(f)f();}

$('#veil').onclick=shut;$$('.x').forEach(b=>b.onclick=shut);

  S.goal=v;S.goals.push({t:v,a:addr('Gc',1)});goalList();save();shut();setLoc(v.slice(0,42),'Gc');};

function bindTog(sel,key,ord){const t=$(sel);if(!t)return;
  t.classList.toggle('on',!!S[key]);
  t.onclick=()=>{S[key]=S[key]?0:1;t.classList.toggle('on',!!S[key]);
    (key==='web'?S.webLog:S.termLog).push({c:'capability '+key+' = '+(S[key]?'allow':'deny'),
      o:'',a:addr('Gf',ord)});save();key==='web'?wbRender():tmRender();};}
function tmRender(){bindTog('#tgTerm','term',1);bindTog('#tgShot','shot',2);

    '<div><span class="c">$ '+esc(l.c)+'</span>'+(l.o?'\n<span class="o">'+esc(l.o)+'</span>':'')+

    :'<div class="none" style="padding:22px">no commands</div>';}
function wbRender(){bindTog('#tgWeb','web',3);
  $('#webLog').innerHTML=S.webLog.length?S.webLog.slice().reverse().map(l=>
    '<div><span class="c">'+esc(l.c)+'</span><div class="ts">'+l.a+'</div></div>').join('')
    :'<div class="none" style="padding:22px">no fetches</div>';}

let shotBase=null;
$('#shotBtn').onclick=async()=>{
  if(!S.shot){$('#shotBtn').textContent='Enable screen capture above first';
    setTimeout(()=>$('#shotBtn').textContent='Take screenshot now',1800);return;}
  try{const st=await navigator.mediaDevices.getDisplayMedia({video:true});
    const v=document.createElement('video');v.srcObject=st;await v.play();
    await new Promise(r=>setTimeout(r,320));
    const c=$('#shotC');c.width=v.videoWidth;c.height=v.videoHeight;
    const x=c.getContext('2d');x.drawImage(v,0,0);st.getTracks().forEach(t=>t.stop());
    shotBase=x.getImageData(0,0,c.width,c.height);
    c.style.display='block';$('#shotTools').style.display='flex';
    S.termLog.push({c:'screen capture',o:c.width+'x'+c.height,a:addr('Gf',4)});
    tmRender();save();setLoc('captured screen','Gf');
  }catch(e){S.termLog.push({c:'screen capture',o:'refused: '+e.message,a:addr('Gf',4)});
    tmRender();save();}};

/* draw on the capture */
(()=>{const c=$('#shotC');let drawing=false;
  const pt=e=>{const r=c.getBoundingClientRect();
    return[(e.clientX-r.left)*(c.width/r.width),(e.clientY-r.top)*(c.height/r.height)];};
  c.addEventListener('pointerdown',e=>{if(!shotBase)return;drawing=true;c.setPointerCapture(e.pointerId);
    const x=c.getContext('2d');const[a,b]=pt(e);
    x.strokeStyle='#a374f5';x.lineWidth=Math.max(3,c.width/220);
    x.lineCap='round';x.lineJoin='round';x.shadowColor='#a374f5';x.shadowBlur=12;
    x.beginPath();x.moveTo(a,b);});
  c.addEventListener('pointermove',e=>{if(!drawing)return;
    const x=c.getContext('2d');const[a,b]=pt(e);x.lineTo(a,b);x.stroke();});
  c.addEventListener('pointerup',()=>{drawing=false;});
  c.addEventListener('pointercancel',()=>{drawing=false;});})();

  $('#pBody').innerHTML='<div style="margin-bottom:13px">'+(b.tag?'<span class="pill">SHORT-TERM</span>':'')+
    '</div>'+Q13.map((q,i)=>'<div class="qa"><div class="q">'+(i+1)+'. '+q+'</div><div class="a">'+
    esc(b.ans&&b.ans[i]||'—')+'</div></div>').join('')+
    '';

const SR=window.SpeechRecognition||window.webkitSpeechRecognition;
$('#mic').onclick=()=>{
  if(!SR){setLoc('voice unavailable here','');return;}
  if(micOn){micOn=0;if(rec)rec.stop();$('#mic').classList.remove('on');setLoc('','');return;}
  rec=new SR();rec.continuous=true;rec.interimResults=false;rec.lang='en-US';
  rec.onresult=ev=>{const t=ev.results[ev.results.length-1][0].transcript.trim();
    if(/shakti/i.test(t)){const m={c:'me',t:t,a:addr('Gh',2)};S.msgs.push(m);paint(m);save();
      setLoc('woken by voice','Gh');}};
  rec.onerror=e=>{setLoc('voice error: '+e.error,'');micOn=0;$('#mic').classList.remove('on');};
  rec.onend=()=>{if(micOn){try{rec.start();}catch(e){}}};
  try{rec.start();micOn=1;$('#mic').classList.add('on');setLoc('listening for her name','Gh');}
  catch(e){setLoc('mic refused','');}};
